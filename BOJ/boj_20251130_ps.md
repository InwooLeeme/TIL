### 1. [<span style="color: #ec9a00;">G3</span>] [Heights](https://www.acmicpc.net/problem/17236)

- **tag** : math, geometry

헤론 정리를 쓰자

```cpp
using ld = long double;
ld ha, hb,hc;

void solve(){
	ri(ha, hb, hc);
	ld A = 1 / ha;
	ld B = 1 / hb;
	ld C = 1 / hc;
	ld S = A + B + C;
	ld area = 1 / sqrt(S * (S - A * 2) * (S - B * 2) * (S - C * 2));
	po(area);
}
```

### 2. [<span style="color: #ec9a00;">G3</span>] [약수 게임](https://www.acmicpc.net/problem/16894)

- **tag** : math, number_theory, game_theory

두 소수의 곱이면 후공승, 이외엔 선공승

```cpp
int n;

void solve(){
	ri(n);
	if(n == 1){
		po("koosaga");
		return;
	}
	vi res;
	for(int i = 2; i *i <= n; i++){
		while(n % i == 0) n /= i, res.pb(i);
	}
	if(n > 1) res.pb(n);
	if(si(res) != 2) po("koosaga");
	else po("cubelover");
}
```

### 3. [<span style="color: #ec9a00;">G5</span>] [최대 합 순서쌍의 개수](https://www.acmicpc.net/problem/30646)

- **tag** : set, prefix_sum, hash_set

인덱스 저장하고 map으로 정리

```cpp
int n;
map<int, vi> M;

void solve(){
	ri(n);
	vi v(n + 1);
	for(int i = 1; i <= n; i++){
		ri(v[i]);
		M[v[i]].pb(i);
		v[i] += v[i - 1];
	}
	int mx = 0, cnt = 1;
	for(const auto& [a, b] : M){
		int L = *b.begin(),R = *prev(b.end());
		int S = v[R] - v[L - 1];
		if(S > mx) mx = S,cnt = 1;
		else if(S == mx) cnt++;
	}
	po(mx, cnt);
}
```

### 4. [<span style="color: #ec9a00;">G4</span>] [수열의 점수](https://www.acmicpc.net/problem/2036)

- **tag** : greedy

0, 1, 양수, 음수 이렇게 4개로 나눠서 처리하면 된다. 음수는 2개씩 짝지어서 처리하자.

```cpp
ll n,res;

void solve(){
	ri(n);
	minpq<ll> mi;
	maxpq<ll> pl;
	for(int i = 0; i < n; i++){
		int x; ri(x);
		if(x == 1) res += x;
		else{
			if(x > 1) pl.push(x);
			else if(x <= 0) mi.push(x);
		}
	}
	while(si(pl) > 1){
		auto a = pl.top(); pl.pop();
		auto b = pl.top(); pl.pop();
		res += (a * b);
	}
	while(si(mi) > 1){
		auto a = mi.top(); mi.pop();
		auto b = mi.top(); mi.pop();
		res += (a * b);
	}
	while(!mi.empty()){
		res += mi.top(); mi.pop();
	}
	while(!pl.empty()){
		res += pl.top(); pl.pop();
	}
	po(res);
}
```

### 5. [<span style="color: #ec9a00;">G5</span>] [십자 찾기](https://www.acmicpc.net/problem/32715)

- **tag** : prefix_sum

각 4방향에 대해서 누적합처리

```cpp
int n,m,k,res, v[2555][2555][4];

void solve(){
	ri(n, m, k);
	for(int i = 1; i <= n; i++) for(int j = 1; j <= m; j++){
		ri(v[i][j][0]);
		v[i][j][3] = v[i][j][2] = v[i][j][1] = v[i][j][0];
	}
	for(int i = 1; i <= n; i++) for(int j = 1; j <= m; j++){
		if(v[i][j][0]) v[i][j][0] += v[i][j - 1][0];
	}
	for(int i = 1; i <= n; i++) for(int j = 1; j <= m; j++){
		if(v[i][j][1]) v[i][j][1] += v[i - 1][j][1];
	}
	for(int i = n; i > 0; i--) for(int j = m; j > 0; j--){
		if(v[i][j][2]) v[i][j][2] += v[i][j + 1][2];
	}
	for(int i = n; i > 0; i--) for(int j = m; j > 0; j--){
		if(v[i][j][3]) v[i][j][3] += v[i + 1][j][3];
		if(v[i][j][0] > k and v[i][j][1] > k and v[i][j][2] > k and v[i][j][3] > k) res++;
	}
	po(res);
}
```

### 7. [<span style="color: #ec9a00;">G5</span>] [직사각형의 개수](https://www.acmicpc.net/problem/2171)

```cpp
int n,res;
set<pii> s;
pii v[5555];

void solve(){
	ri(n);
	for(int i = 0; i < n; i++){
		ri(v[i].fi, v[i].se);
		s.insert(v[i]);
	}

		}
		s.erase(v[i]);
	}
	po(res);
}
```

### 8. [<span style="color: #ec9a00;">G4</span>] [점프](https://www.acmicpc.net/problem/2253)

- **tag** : bfs

BFS를 잘 짜자

```cpp
int n,m,g[10101];
bool bound(int x){ return x < 0 or x > n; }
int D[3] = {-1, 0, 1};

void solve(){
	ri(n, m);
	while(m--){
		int x; ri(x); g[x] = 1;
	}
	vvi dist(n + 1, vi(500, -1));
	queue<pii> q;
	dist[1][0] = 0; q.push({0, 1}); // 몇번 이동, 위치
	while(si(q)){
		auto [cnt, cur] = q.front(); q.pop();
		if(cur == n){
			po(dist[cur][cnt]);
			return;
		}
		for(int i = 0; i < 3; i++){
			auto ncnt = cnt + D[i];
			auto nxt = cur + ncnt;
			if(ncnt < 1) continue;
			if(bound(nxt)) continue;
			if(g[nxt]) continue;
			if(dist[nxt][ncnt] != -1) continue;
			dist[nxt][ncnt] = dist[cur][cnt] + 1; q.push({ncnt, nxt});
		}
	}
	po(-1);
}
```

### 10. [<span style="color: #ec9a00;">G2</span>] [게임](https://www.acmicpc.net/problem/1103)

- **tag** : dp, graph, dfs

사이클 찾고 최장거리 DP

```cpp
int n,m,mx;
bool bound(int x, int y){ return x < 0 or x >= n or y < 0 or  y >= m; }
string v[55];
bool vist[55][55], finished[55][55];
bool cycle = 0;
int D[55][55];

void dfs(int x, int y){
	vist[x][y] = 1;
	for(int d = 0; d < 4; d++){
		int k = v[x][y] - '0';
		int nx = x + k * dx[d], ny = y + k * dy[d];
		if(bound(nx, ny) or v[nx][ny] == 'H') continue;
		if(vist[nx][ny]){
			if(!finished[nx][ny]) cycle = 1;
		}
		else dfs(nx, ny);
	}
	finished[x][y] = 1;
}

int dp(int x, int y){
	int& ret = D[x][y];
	if(ret != -1) return ret;
	ret = 1;
	int k = v[x][y] - '0';
	for(int d = 0; d < 4; d++){
		int nx = x + k * dx[d], ny = y + k * dy[d];
		if(bound(nx, ny) or v[nx][ny] == 'H') continue;
		ret = max<int>(ret, dp(nx, ny) + 1);
	}
	return ret;
}

void solve(){
	ri(n, m);
	for(int i = 0; i < n; i++) ri(v[i]);
	dfs(0, 0);
	if(cycle){
		po(-1);
		return;
	}
	memset(D, -1, sizeof(D));
	po(dp(0, 0));
}
```

### 11. [<span style="color: #ec9a00;">G2</span>] [로고](https://www.acmicpc.net/problem/3108)

- **tag** : disjoint_set, geometry

좌표축을 옮겨서 처리하면 구현이 쉽다..

```cpp
int n,res;
int v[2222][2222];
int dist[2222][2222];
bool bound(int x, int y){ return x < 0 or x > 2000 or y < 0 or y > 2000; }

void bfs(int x, int y){
	queue<pii> q;
	dist[x][y] = 0; v[x][y] = 0; q.push({x, y});
	while(si(q)){
		auto [x, y] = q.front(); q.pop();
		v[x][y] = 0;
		for(int d = 0; d < 4; d++){
			int nx = x + dx[d], ny = y + dy[d];
			if(bound(nx, ny) or !v[nx][ny]) continue;
			if(dist[nx][ny] != -1) continue;
			dist[nx][ny] = dist[x][y] + 1; q.push({nx, ny});
		}
	}
}

void solve(){
	ri(n);
	memset(dist, -1, sizeof(dist));
	for(int i = 1; i <= n; i++){
		int x1, y1, x2, y2; ri(x1, y1, x2, y2);
		if(x1 > x2) swap(x1, x2);
		if(y1 > y2) swap(y1, y2);
		x1 = 2 * x1 + 1000; y1 = 2 * y1 + 1000;
		x2 = 2 * x2 + 1000; y2 = 2 * y2 + 1000;
		for(int r = x1; r <= x2; r++) v[r][y1] = v[r][y2] = i;
		for(int c = y1; c <= y2; c++) v[x1][c] = v[x2][c] = i;
	}
	if(v[1000][1000]) res--;
	for(int i = 0; i <= 2000; i++) for(int j = 0; j <= 2000; j++){
		if(v[i][j] and dist[i][j] == -1){
			res++;
			bfs(i, j);
		}
	}
	po(res);
}
```

### 12. [<span style="color: #ec9a00;">G5</span>] [Binary Roads](https://www.acmicpc.net/problem/13213)

- **tag** : graph, bfs

```cpp
int32_t n,m;
vector<pair<int32_t,int32_t>> g[202020];

void solve(){
	ri(n, m);
	for(int i = 0; i < m; i++){
		int a,b,c; ri(a, b, c);
		g[a].pb({c, b}); g[b].pb({c, a});
	}
	queue<pii> q;
	vector<vector<int32_t>> dist(n + 1, vector<int32_t>(2, -1));
	dist[0][0] = 0; dist[0][1] = 0; q.push({0, 0}); q.push({1, 0});
	while(si(q)){
		auto [T, cur] = q.front(); q.pop();
		if(cur == n - 1){
			po(dist[cur][T]);
			return;
		}
		for(const auto& [c, nxt] : g[cur]){
			if(T != c) continue;
			auto nT = T ^ 1;
			if(dist[nxt][nT] != -1) continue;
			dist[nxt][nT] = dist[cur][T] + 1; q.push({nT, nxt});
		}
	}
	po(-1);
}
```

### 13. [<span style="color: #ec9a00;">G4</span>] [Debug](https://www.acmicpc.net/problem/15244)

- **tag** : prefix_sum

각 $Xi$값들을 전부 모아서 처리하면 시간이 줄어들 것 같다.
정렬한 다음에 모아준 값을 들고 누적합을 만들 다음에 $Q$개의 쿼리를 처리하자.

```cpp
int n,m, q, v[1 << 20], p[1 << 20];

void solve(){
	ri(n, m);
	map<int, int> M;
	for(int i = 0; i < m; i++){
		int x; ri(x); M[x]++;
	}
	for(const auto& [a, b] : M){
		for(int i = 1; i <= n; i += a) v[i] += b;
	}
	for(int i = 1; i <= n; i++) p[i] = p[i - 1] + v[i];
	ri(q);
	while(q--){
		int l,r; ri(l, r); ++l,++r;
		po(p[r] - p[l - 1]);
	}
}
```

### 15. [<span style="color: #ec9a00;">G4</span>] [퍼레이드](https://www.acmicpc.net/problem/16168)

- **tag** : graph theory

문제에서 설명하는 경로가 오일러 경로(한붓 그리기)와 동일하다.

```cpp
int n, m, deg[3030];
vi g[3030];
bitset<3030> vist;

void dfs(int cur){
	vist[cur] = 1;
	for(const auto& nxt : g[cur]){
		if(vist[nxt]) continue;
		dfs(nxt);
	}
}

void solve(){
	ri(n, m);
	for(int i = 0; i < m; i++){
		int a, b; ri(a, b);
		g[a].pb(b); g[b].pb(a);
		deg[a]++; deg[b]++;
	}
	int components = 0;
	for(int i = 1; i <= n; i++){
		if(vist[i]) continue;
		components++; dfs(i);
	}
	if(components >= 2){
		po("NO");
		return;
	}
	int odd = 0;
	for(int i = 1; i <= n; i++){
		odd += (deg[i] & 1);
	}
	if(odd == 0 or odd == 2) po("YES");
	else po("NO");
}
```

### 16. [<span style="color: #ec9a00;">G4</span>] [개업 2](https://www.acmicpc.net/problem/13902)

- **tag** : bfs, dp

bfs로도 된다.

```cpp
int n,m;
int dist[10101];

void solve(){
	memset(dist, -1, sizeof(dist));
	ri(n, m);
	vi v(m); ri(v);
	vi w = v;
	for(int i = 0; i < m; i++) for(int j = i + 1; j < m; j++) w.pb(v[i] + v[j]);

		auto cur = q.front(); q.pop();

			return;
		}
		for(const auto& i : w){
			auto nxt = i + cur;
			if(nxt > n or dist[nxt] != -1) continue;
			dist[nxt] = dist[cur] + 1; q.push(nxt);
		}
	}
	po(-1);
}
```

### 17. [<span style="color: #ec9a00;">G3</span>] [곰팡이](https://www.acmicpc.net/problem/1888)

- **tag** : simulation, bfs

구현 근육을 테스트해보자. 문제에 있는 그대로 구현해주면 된다.

```cpp
int n,m;
int v[111][111], w[111][111];
bool vist[111][111];
bool bound(int x, int y){ return x < 0 or x >= n or y < 0 or y >= m; }

void dfs(int x, int y){
	vist[x][y] = 1;
	for(int d = 0; d < 4; d++){
		auto nx = x + dx[d], ny = y + dy[d];
		if(bound(nx, ny)) continue;
		if(vist[nx][ny] or !v[nx][ny]) continue;
		dfs(nx, ny);
	}
}

void solve(){
	ri(n, m);
	int res = 0,cnt = 0;
	for(int i = 0; i < n; i++){
		string s; ri(s);
		for(int j = 0; j < m; j++){
			if(s[j] > '0') v[i][j] = s[j] - '0', w[i][j] = v[i][j],cnt++;
			else v[i][j] = s[j] - '0', w[i][j] = v[i][j];
		}
	}
	if(!cnt){
		po(0);
		return;
	}
	while(1){
		memset(vist, 0, sizeof(vist));
		cnt = 0;
		queue<ti3> q;
		for(int i = 0; i < n; i++) for(int j = 0; j < m; j++){
			w[i][j] = v[i][j];
			if(v[i][j] and !vist[i][j]){
				dfs(i, j); cnt++;
			}
		}
		if(cnt == 1) break;
		for(int i = 0; i < n; i++) for(int j = 0; j < m; j++){
			if(w[i][j]){
				q.push({w[i][j], i, j});
			}
		}
		while(si(q)){
			auto [K, x, y] = q.front(); q.pop();
			for(int nx = x - K; nx <= x + K; nx++){
				for(int ny = y - K; ny <= y + K; ny++){
					if(bound(nx, ny)) continue;
					w[nx][ny] = max<int>(w[nx][ny], K);
				}
			}
		}
		for(int i = 0; i < n; i++) for(int j = 0; j < m; j++) v[i][j] = w[i][j];
		res++;
	}
	po(res);
}

```

### 18. [<span style="color: #ec9a00;">G2</span>] [Labyrinth](https://www.acmicpc.net/problem/3482)

- **tag** : dfs

트리의 지름은 dfs 2번으로 구할 수 있음이 잘 알려져있다.
이걸 2차원에서 해주자.

```cpp
int vist[1111][1111];

void solve(){
	int t; ri(t); while(t--){
		int N, M; ri(M, N);
		vs v(N); ri(v);
		int mx = -1, r = -1, c = -1;
		memset(vist, 0, sizeof(vist));
		auto dfs = [&](int x, int y, int len, auto&& dfs) -> void{
			if(len > mx){
				mx = len; r = x, c = y;
			}
			vist[x][y] = 1;
			for(int d = 0; d < 4; d++){
				auto nx = x + dx[d], ny = y + dy[d];
				if(nx < 0 or nx >= N or ny < 0 or ny >= M) continue;
				if(vist[nx][ny] or v[nx][ny] == '#') continue;
				dfs(nx, ny, len + 1, dfs);
			}
		};
		[&](){
			for(int i = 0; i < N; i++) for(int j = 0; j < M; j++){
				if(v[i][j] == '#') continue;
				dfs(i, j, 0, dfs);
				return;
			}
		}();
		memset(vist, 0, sizeof(vist));
		dfs(r, c, 0, dfs);
		cout << "Maximum rope length is " << mx << ".\n";
	}
}
```

### 19. [<span style="color: #27e2a4;">P3</span>] [プラグ (Plugs)](https://www.acmicpc.net/problem/24126)

- **tag** : prefix_sum, bipartite_matching

간선을 단순하게 이어준다면 시간초과다.
X축을 플러그 Y축을 콘센트로 놓고 2차원으로 생각한다면 imos법을 이용해서 $O(N^2)$에 서로 연결할 수 없는 걸 확인할 수 있을 것이고 이걸 바탕으로 간선을 연결해준 다음에 최대 매칭을 구해주면 된다.

```cpp
int n,m, v[3333][3333], P[3333][3333];
int res[3001];

// n: 왼쪽 정점 개수, m: 오른쪽 정점 개수, 0-based
struct HopcroftKarp{
  int n, m;
  vector<vector<int>> g;
  vector<int> dst, le, ri;
  vector<char> visit, track;
  HopcroftKarp(int n, int m) : n(n), m(m) { clear(); }
  void clear(){
    g = vector<vector<int>>(n); dst = vector<int>(n, 0);
    le = vector<int>(n, -1); ri = vector<int>(m, -1);
    visit = vector<char>(n, 0); track = vector<char>(n+m, 0);
  }
  void add_edge(int s, int e){ g[s].push_back(e); }
  bool bfs(){
    bool res = false;
    queue<int> que;
    fill(dst.begin(), dst.end(), 0);
    for(int i=0; i<n; i++) if(le[i] == -1) que.push(i), dst[i] = 1;
    while(!que.empty()){
      int v = que.front(); que.pop();
      for(auto i : g[v]){
        if(ri[i] == -1) res = true;
        else if(!dst[ri[i]]) dst[ri[i]] = dst[v] + 1, que.push(ri[i]);
      }
    }
    return res;
  }
  bool dfs(int v){
    if(visit[v]) return false;
    visit[v] = 1;
    for(auto i : g[v]){
      if(ri[i] == -1 || !visit[ri[i]] && dst[ri[i]] == dst[v] + 1 && dfs(ri[i])){
        le[v] = i; ri[i] = v; return true;
      }
    }
    return false;
  }
  int maximum_matching(){
    int res = 0;
    fill(le.begin(), le.end(), -1);
    fill(ri.begin(), ri.end(), -1);
    while(bfs()){
      fill(visit.begin(), visit.end(), 0);
      for(int i=0; i<n; i++) if(le[i] == -1) res += dfs(i);
    }
    return res;
  }
  vector<pair<int,int>> maximum_matching_edges(){
    int matching = maximum_matching();
    vector<pair<int,int>> edges; edges.reserve(matching);
    for(int i=0; i<n; i++) if(le[i] != -1) edges.emplace_back(i, le[i]);
    return edges;
  }
  void dfs_track(int v){
    if(track[v]) return; track[v] = 1;
    for(auto i : g[v]) track[n+i] = 1, dfs_track(ri[i]);
  }
  tuple<vector<int>, vector<int>, int> minimum_vertex_cover(){
    int matching = maximum_matching();
    fill(track.begin(), track.end(), 0);
    for(int i=0; i<n; i++) if(le[i] == -1) dfs_track(i);
    vector<int> lv, rv;
    for(int i=0; i<n; i++) if(!track[i]) lv.push_back(i);
    for(int i=0; i<m; i++) if(track[n+i]) rv.push_back(i);
    assert(lv.size() + rv.size() == matching);
    return {lv, rv, lv.size() + rv.size()};
  }
};

void solve(){
	ri(n, m);
	for(int _ = 0; _ < m; _++){
		int a,b,c,d; ri(a, b, c, d);
		v[a][c]++;
		v[b + 1][c]--;
		v[a][d + 1]--;
		v[b + 1][d + 1]++;
	}
	for(int j = 1; j < 3001; j++) for(int i = 1; i < 3001; i++) P[i][j] = P[i - 1][j] + v[i][j];
	for(int i = 1; i < 3001; i++) for(int j = 1; j < 3001; j++) P[i][j] += P[i][j - 1];
	HopcroftKarp flow(n + 1, n + 1);
	for(int i = 1; i <= n; i++){
		for(int j = 1; j <= n; j++){
			if(P[i][j] <= 0) flow.add_edge(i, j);
		}
	}
	int cnt = flow.maximum_matching();
	for(int i = 1; i <= n; i++){
		po(flow.le[i]);
	}
}
```

### 21. [<span style="color: #27e2a4;">P4</span>] [Wooden Fence](https://www.acmicpc.net/problem/6646)

- **tag** : brute_force, convex_hull

$O(2^N)$가지 경우의 수에 대해서 그냥 전부 다 만들어보면서 찾아주면 된다.

```cpp
const double EPS = 1e-12;
const double PI = acos(-1);

struct point {
	int x, y;
	/* constructors */
	point() : x(0), y(0) {}
	point(int x, int y) : x(x), y(y) {}
	/* methods */
	friend int dist(point a, point b) { // (Euclidean distance)^2, use ll if x, y > 1e4
		return (a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y);
	}
	friend int ccw(point a, point b, point c) {
		int t = (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		return t ? t > 0 ? 1 : -1 : 0;
	}
	friend int is_cross(point a, point b, point c, point d) { // cross : -1(X), intersect : 1(X or L)
		int ab = ccw(a, b, c) * ccw(a, b, d);
		int cd = ccw(c, d, a) * ccw(c, d, b);
		if (ab < 0 && cd < 0) return -1;
		if (ab == 0 && cd == 0) {
			if (a > b) swap(a, b);
			if (c > d) swap(c, d);
			if (!ccw(a, b, c) and !ccw(a, b, d) and !ccw(c, d, a) and !ccw(c, d, b) and c < b and a < d) return 2; // intersect INF
			return c <= b && a <= d;
		}
		return ab <= 0 && cd <= 0;
	}
	friend point findInterCod(point a, point b, point c, point d){
		double x = (a.x*b.y - a.y*b.x)*(c.x - d.x) - (a.x - b.x) * (c.x*d.y - c.y*d.x);
		double y = (a.x*b.y - a.y*b.x)*(c.y - d.y) - (a.y - b.y) * (c.x*d.y - c.y*d.x);
		double div = (a.x - b.x)*(c.y - d.y) - (a.y - b.y) * (c.x - d.x);
		if(div == 0.0){
			if(b == c and a <= c) return b;
			else if(a == d and a >= c) return a;
		}
		return point(x/div, y/div);
	}
	/* operators */
	bool operator == (const point& i) const { return x == i.x && y == i.y; }
	bool operator <  (const point& i) const { return x < i.x || x == i.x && y < i.y; }
	bool operator >  (const point& i) const { return i < *this; }
	bool operator != (const point& i) const { return !(*this == i); }
	bool operator <= (const point& i) const { return *this < i || *this == i; }
	bool operator >= (const point& i) const { return *this > i || *this == i; }
	friend int operator * (point a, point b) { return a.x * b.y - a.y * b.x; } // cross product
	friend point operator + (point a, point b) { return { a.x + b.x, a.y + b.y }; }
	friend point operator - (point a, point b) { return { a.x - b.x, a.y - b.y }; }
	friend istream& operator >> (istream& in, point& i) { in >> i.x >> i.y; return in; }
};

struct line {
	point a, b;
	/* constructors */
	line() {}
	line(point a, point b) : a(a), b(b) { if (a > b) swap(a, b); }
	/* methods */
	double length() { return sqrt(dist(a, b)); }
	bool is_on_line(point c) { return !ccw(a, b, c) && min(a, b) <= c && c <= max(a, b); }
	friend int is_cross(line a, line b) { return is_cross(a.a, a.b, b.a, b.b); }
	// point to line distance
	friend double dist(line a, point b) {
		auto CCW = [&](point& a, point& b, point& c) -> int {
		    return (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		};
		return CCW(a.a, a.b, b) / a.length();
	}
	/* operators */
	friend istream& operator >> (istream& in, line& i) { in >> i.a >> i.b; return in; }
};

struct polygon {
	vector<point> v;
	/* constructors */
	polygon() {}
	polygon(vector<point> v) : v(v) {}
	/* methods */
	void convex_hull() { // convex hull in O(nlogn), ccw(v[i], v[i + 1], v[i + 2]) == 1
		if (v.size() < 3) return;
		swap(v[0], *min_element(v.begin(), v.end()));
		sort(v.begin() + 1, v.end(), [&](const point& a, const point& b) {
			int cw = ccw(v[0], a, b);
			if (cw) return cw > 0;
			return dist(v[0], a) < dist(v[0], b);
		});
		vector<point> res;
		for (const auto& i : v) {
			while (res.size() >= 2 && ccw(res[res.size() - 2], res.back(), i) <= 0) res.pop_back();
			res.push_back(i);
		}
		v = res;
	}
	double rotating_calipers() { // maximum distance in O(n), must be convex polygon
        if (v.size() < 2) return 0;
		int n = v.size();
		int l = 0, r = max_element(v.begin(), v.end()) - v.begin();
		double ret = sqrt(dist(v[l], v[r]));
		for (int i = 0; i < n; i++) {
			if ((v[(l + 1) % n] - v[l]) * (v[(r + 1) % n] - v[r]) < 0) l = (l + 1) % n;
			else r = (r + 1) % n;
			ret = max(ret, sqrtl(dist(v[l], v[r])));
		}
		return ret;
	}
	int point_in_polygon_naive(point a) { // O(qn), for all polygon, inside : 1, boundary : -1
		if (v.size() < 3) return 0;
		line cur(a, { int(1e9), a.y + 1 });
		int cnt = 0;
		for (int i = 0; i < v.size(); i++) {
			line l(v[i], v[(i + 1) % v.size()]);
			if (is_cross(cur, l) == -1) cnt++;
			if (l.is_on_line(a)) return -1;
		}
		return cnt % 2;
	}
	int point_in_polygon(point a) { // O(qlogn), for convex polygon only
		if (v.size() < 3) return 0;
		if (a < v[0] || ccw(v[0], v[1], a) < 0 || ccw(v[0], v.back(), a) > 0) return 0;
		if (ccw(v[0], v[1], a) == 0) return line(v[0], v[1]).is_on_line(a) ? -1 : 0;
		if (ccw(v[0], v.back(), a) == 0) return line(v[0], v.back()).is_on_line(a) ? -1 : 0;
		int lo = 0, hi = v.size();
		while (lo + 1 < hi) {
			int mid = lo + hi >> 1;
			if (ccw(v[0], v[mid], a) > 0) lo = mid;
			else hi = mid;
		}
		int cw = ccw(v[lo], v[lo + 1], a);
		return cw ? cw > 0 ? 1 : 0 : -1;
	}
	double area() {
		auto CCW = [&](point& a, point& b, point& c) -> int {
		    return (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		};
		int ret = 0;
		for (int i = 1; i < v.size() - 1; i++) {
			ret += CCW(v[0], v[i], v[i + 1]);
		}
		return abs(ret) / 2.;
	}
};

void solve(){
	int n; while(cin >> n and n){
		vector<point> v(n);
		vi V(n),L(n);
		for(int i = 0; i < n; i++){
			ri(v[i], V[i], L[i]);
		}
		int mn = INF;
		for(int bit = 0; bit < (1 << n); bit++){
			int S = 0, len = 0;
			vector<point> w;
			for(int j = 0; j < n; j++){
				if(bit & (1 << j)){
					len += L[j]; S += V[j];
				}
				else w.push_back(v[j]);
			}
			polygon p(w); p.convex_hull();
			double curS = 0;
			for(int i = 0; i < si(p.v); i++){
				curS += sqrtl(dist(p.v[i], p.v[(i + 1) % si(p.v)]));
			}
			if(len < curS) continue;
			mn = min<int>(mn, S);
		}
		cout << "The lost value is " << mn << ".\n";
	}
}
```

### 22. [<span style="color: #27e2a4;">P2</span>] [Cow Curling](https://www.acmicpc.net/problem/9875)

```cpp
}
```

### 23. [<span style="color: #27e2a4;">P3</span>] [양궁](https://www.acmicpc.net/problem/27656)

```cpp
const double EPS = 1e-12;
const double PI = acos(-1);

struct point {
	int x, y;
	/* constructors */
	point() : x(0), y(0) {}
	point(int x, int y) : x(x), y(y) {}
	/* methods */
	friend int dist(point a, point b) { // (Euclidean distance)^2, use ll if x, y > 1e4
		return (a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y);
	}
	friend int ccw(point a, point b, point c) {
		int t = (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		return t ? t > 0 ? 1 : -1 : 0;
	}
	friend int is_cross(point a, point b, point c, point d) { // cross : -1(X), intersect : 1(X or L)
		int ab = ccw(a, b, c) * ccw(a, b, d);
		int cd = ccw(c, d, a) * ccw(c, d, b);
		if (ab < 0 && cd < 0) return -1;
		if (ab == 0 && cd == 0) {
			if (a > b) swap(a, b);
			if (c > d) swap(c, d);
			if (!ccw(a, b, c) and !ccw(a, b, d) and !ccw(c, d, a) and !ccw(c, d, b) and c < b and a < d) return 2; // intersect INF
			return c <= b && a <= d;
		}
		return ab <= 0 && cd <= 0;
	}
	friend point findInterCod(point a, point b, point c, point d){
		double x = (a.x*b.y - a.y*b.x)*(c.x - d.x) - (a.x - b.x) * (c.x*d.y - c.y*d.x);
		double y = (a.x*b.y - a.y*b.x)*(c.y - d.y) - (a.y - b.y) * (c.x*d.y - c.y*d.x);
		double div = (a.x - b.x)*(c.y - d.y) - (a.y - b.y) * (c.x - d.x);
		if(div == 0.0){
			if(b == c and a <= c) return b;
			else if(a == d and a >= c) return a;
		}
		return point(x/div, y/div);
	}
	/* operators */
	bool operator == (const point& i) const { return x == i.x && y == i.y; }
	bool operator <  (const point& i) const { return x < i.x || x == i.x && y < i.y; }
	bool operator >  (const point& i) const { return i < *this; }
	bool operator != (const point& i) const { return !(*this == i); }
	bool operator <= (const point& i) const { return *this < i || *this == i; }
	bool operator >= (const point& i) const { return *this > i || *this == i; }
	friend int operator * (point a, point b) { return a.x * b.y - a.y * b.x; } // cross product
	friend point operator + (point a, point b) { return { a.x + b.x, a.y + b.y }; }
	friend point operator - (point a, point b) { return { a.x - b.x, a.y - b.y }; }
	friend istream& operator >> (istream& in, point& i) { in >> i.x >> i.y; return in; }
};

struct line {
	point a, b;
	/* constructors */
	line() {}
	line(point a, point b) : a(a), b(b) { if (a > b) swap(a, b); }
	/* methods */
	double length() { return sqrt(dist(a, b)); }
	bool is_on_line(point c) { return !ccw(a, b, c) && min(a, b) <= c && c <= max(a, b); }
	friend int is_cross(line a, line b) { return is_cross(a.a, a.b, b.a, b.b); }
	// point to line distance
	friend double dist(line a, point b) {
		auto CCW = [&](point& a, point& b, point& c) -> int {
		    return (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		};
		return CCW(a.a, a.b, b) / a.length();
	}
	/* operators */
	friend istream& operator >> (istream& in, line& i) { in >> i.a >> i.b; return in; }
};

struct polygon {
	vector<point> v;
	/* constructors */
	polygon() {}
	polygon(vector<point> v) : v(v) {}
	/* methods */
	void convex_hull() { // convex hull in O(nlogn), ccw(v[i], v[i + 1], v[i + 2]) == 1
		if (v.size() < 3) return;
		swap(v[0], *min_element(v.begin(), v.end()));
		sort(v.begin() + 1, v.end(), [&](const point& a, const point& b) {
			int cw = ccw(v[0], a, b);
			if (cw) return cw > 0;
			return dist(v[0], a) < dist(v[0], b);
		});
		vector<point> res;
		for (const auto& i : v) {
			while (res.size() >= 2 && ccw(res[res.size() - 2], res.back(), i) <= 0) res.pop_back();
			res.push_back(i);
		}
		v = res;
	}
	double rotating_calipers() { // maximum distance in O(n), must be convex polygon
        if (v.size() < 2) return 0;
		int n = v.size();
		int l = 0, r = max_element(v.begin(), v.end()) - v.begin();
		double ret = sqrt(dist(v[l], v[r]));
		for (int i = 0; i < n; i++) {
			if ((v[(l + 1) % n] - v[l]) * (v[(r + 1) % n] - v[r]) < 0) l = (l + 1) % n;
			else r = (r + 1) % n;
			ret = max(ret, sqrtl(dist(v[l], v[r])));
		}
		return ret;
	}
	int point_in_polygon_naive(point a) { // O(qn), for all polygon, inside : 1, boundary : -1
		if (v.size() < 3) return 0;
		line cur(a, { int(1e9), a.y + 1 });
		int cnt = 0;
		for (int i = 0; i < v.size(); i++) {
			line l(v[i], v[(i + 1) % v.size()]);
			if (is_cross(cur, l) == -1) cnt++;
			if (l.is_on_line(a)) return -1;
		}
		return cnt % 2;
	}
	int point_in_polygon(point a) { // O(qlogn), for convex polygon only
		if (v.size() < 3) return 0;
		if (a < v[0] || ccw(v[0], v[1], a) < 0 || ccw(v[0], v.back(), a) > 0) return 0;
		if (ccw(v[0], v[1], a) == 0) return line(v[0], v[1]).is_on_line(a) ? -1 : 0;
		if (ccw(v[0], v.back(), a) == 0) return line(v[0], v.back()).is_on_line(a) ? -1 : 0;
		int lo = 0, hi = v.size();
		while (lo + 1 < hi) {
			int mid = lo + hi >> 1;
			if (ccw(v[0], v[mid], a) > 0) lo = mid;
			else hi = mid;
		}
		int cw = ccw(v[lo], v[lo + 1], a);
		return cw ? cw > 0 ? 1 : 0 : -1;
	}
	double area() {
		auto CCW = [&](point& a, point& b, point& c) -> int {
		    return (b.x - a.x) * (c.y - b.y) - (c.x - b.x) * (b.y - a.y);
		};
		int ret = 0;
		for (int i = 1; i < v.size() - 1; i++) {
			ret += CCW(v[0], v[i], v[i + 1]);
		}
		return abs(ret) / 2.;
	}
};


int n;

void solve(){
	ri(n);
	vector<point> v(n); ri(v);
	set<point> M;
	vector<polygon> poly;
	while(si(M) < n){
		vector<point> w; for(int i = 0; i < n; i++) if(!M.count(v[i])) w.push_back(v[i]);
		if(w.empty()) break;
		polygon p(w); p.convex_hull();
		for(const auto& i : p.v) M.insert(i);
		poly.pb(p);
	}
	auto check = [&](const int mid, point x) -> bool{
		return poly[mid].point_in_polygon(x) ? 0 : 1;
	};
	int q; ri(q); while(q--){
		point x; ri(x);
		int l = -1, r = si(poly);
		while(l + 1 < r){
			int mid = l + r >> 1;
			if(!check(mid, x)) l = mid;
			else r = mid;
		}
		po(r);
	}
}
```
